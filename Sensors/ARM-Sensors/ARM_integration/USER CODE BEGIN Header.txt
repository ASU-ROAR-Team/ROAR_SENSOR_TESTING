void motor_move(uint8_t speed, MotorConfig *m) {
	Motor_Direction direction;
	uint8_t duty;

	if (speed == 64) {
		direction = MOTOR_STOP;
		duty = 0;
	} else if (speed < 64) {
		direction = MOTOR_REVERSE;
		duty = (uint8_t) ((63 - speed) * 100 / 63);
	} else {
		direction = MOTOR_FORWARD;
		duty = (uint8_t) ((speed - 65) * 100 / 62);
	}

// Set direction pin
	switch (direction) {
	case MOTOR_FORWARD:
		HAL_GPIO_WritePin(m->dir_port, m->dir_pin, GPIO_PIN_SET);
		break;
	case MOTOR_REVERSE:
		HAL_GPIO_WritePin(m->dir_port, m->dir_pin, GPIO_PIN_RESET);
		break;
	case MOTOR_STOP:
		HAL_GPIO_WritePin(m->dir_port, m->dir_pin, GPIO_PIN_RESET);
		break;
	}

// Set PWM duty
	uint32_t max_pulse = m->htim->Instance->ARR;
	uint32_t pulse_value = (uint32_t) (((float) duty / 100.0f)
			* (float) max_pulse);
	__HAL_TIM_SET_COMPARE(m->htim, m->channel, pulse_value);
}

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
	if (huart->Instance == USART3) {
		for (uint8_t i = 0; i < NUM_MOTORS; i++) {
			speeds[i] = rx_buffer[i];

			if (speeds[i] > 127)
				speeds[i] = 127;
		}
		for (int i = 0; i < 6; i++) {
			current_speed[i] = (speeds[i] - last_speeds[i]);
		}
		if (current_speed[0] > 0) {
			for (int i = 0; i <= current_speed[0]; i++) {
				motor_move(last_speeds[0] + i, &motors[0]);
			}
		} else {
			for (int i = 0; i <= abs(current_speed[0]); i++) {
				motor_move(last_speeds[0] - i, &motors[0]);
			}
		}
		//MOTOR1
		if (current_speed[1] > 0) {
			for (int i = 0; i <= current_speed[1]; i++) {
				motor_move(last_speeds[1] + i, &motors[1]);
			}
		} else {
			for (int i = 0; i <= abs(current_speed[1]); i++) {
				motor_move(last_speeds[1] - i, &motors[1]);
			}
		}
		if (current_speed[2] > 0) {
			for (int i = 0; i <= current_speed[2]; i++) {
				motor_move(last_speeds[2] + i, &motors[2]);
			}
		} else {
			for (int i = 0; i <= abs(current_speed[2]); i++) {
				motor_move(last_speeds[2] - i, &motors[2]);
			}
		}
		if (current_speed[3] > 0) {
			for (int i = 0; i <= current_speed[3]; i++) {
				motor_move(last_speeds[3] + i, &motors[3]);
			}
		} else {
			for (int i = 0; i <= abs(current_speed[3]); i++) {
				motor_move(last_speeds[3] - i, &motors[3]);
			}
		}
		if (current_speed[5] > 0) {
			for (int i = 0; i <= current_speed[5]; i++) {
				motor_move(last_speeds[5] + i, &motors[4]);
			}
		} else {
			for (int i = 0; i <= abs(current_speed[5]); i++) {
				motor_move(last_speeds[5] - i, &motors[4]);
			}
		}
		if (current_speed[4] > 0) {
			for (int i = 0; i <= current_speed[4]; i++) {
				motor_move(last_speeds[4] + i, &motors[5]);
			}
		} else {
			for (int i = 0; i <= abs(current_speed[4]); i++) {
				motor_move(last_speeds[4] - i, &motors[5]);
			}
		}
		for (int i = 0; i < 6; i++) {
			last_speeds[i] = speeds[i];
		}
// Restart reception

		HAL_UART_Receive_IT(&huart3, rx_buffer, NUM_MOTORS);
	}
}
